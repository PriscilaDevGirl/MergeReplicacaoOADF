  370 BASE         // Perform a deep comparison to check if two objects are equal.
  371 BASE         _.isEqual = function(a, b) {
  372 BASE           // Check object identity.
  373 BASE           if (a === b) return true;
  374 BASE           // Different types?
  375 BASE           var atype = typeof(a), btype = typeof(b);
  376 BASE           if (atype != btype) return false;

  378 BASE           if (a == b) return true;
  379 BASE           // One of them implements an isEqual()?
  380 BASE           if (a.isEqual) return a.isEqual(b);
  381 BASE           // If a is not an object by this point, we can't handle it.
  382 BASE           if (atype !== 'object') return false;
  383 BASE           // Nothing else worked, deep compare the contents.
  384 BASE           var aKeys = _.keys(a), bKeys = _.keys(b);
  385 BASE           // Different object sizes?
  386 BASE           if (aKeys.length != bKeys.length) return false;
  387 BASE           // Recursive comparison of contents.
  388 BASE           for (var key in a) if (!_.isEqual(a[key], b[key])) return false;
  389 BASE           return true;
